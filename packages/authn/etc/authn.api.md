```ts
import type { KeyAlgorithm, KeyMaterial } from "./keys.js";
/** Canvas C8 D4 codes in check order; a verification reports the first check that fails. */
export type JwsErrorCode = "malformed" | "bad_header" | "unknown_key" | "bad_signature" | "invalid_claims" | "wrong_issuer" | "wrong_audience" | "expired" | "not_yet_valid";
/** Exactly these members, written in this order. */
export interface JwsHeader {
    readonly alg: KeyAlgorithm;
    readonly kid: string;
    readonly typ: string;
}
/** iat and exp are integer Unix seconds. */
export interface JwsClaims {
    readonly iss: string;
    readonly aud: string;
    readonly iat: number;
    readonly exp: number;
}
/** A KeyRecord is one. */
export type JwsKey = KeyMaterial & {
    readonly kid: string;
};
export interface JwsVerifyOptions<C extends JwsClaims, K extends JwsKey = JwsKey> {
    /** Exact match: 'canvas-rt+jwt', 'spinajs-client+jwt'. */
    readonly typ: string;
    /** A header alg outside this list is bad_header. */
    readonly algorithms: readonly KeyAlgorithm[];
    /** Usable keys only (not revoked, not expired); null is unknown_key, and so is a key whose alg differs from the header's. */
    readonly keys: (kid: string) => Promise<K | null>;
    /** The payload rules beyond iss, aud, iat and exp; null is invalid_claims. */
    readonly claims: (payload: Readonly<Record<string, unknown>>) => C | null;
    /** A function when the issuer depends on the key, e.g. an assertion's iss is the key owner. */
    readonly issuer: string | ((iss: string, key: K) => boolean);
    readonly audience: string;
    /** In whole seconds: expired when now ≥ exp + skewSeconds, not_yet_valid when iat > now + skewSeconds. */
    readonly now: Date;
    readonly skewSeconds: number;
    /** exp − iat above this is invalid_claims, as is exp ≤ iat. */
    readonly maxLifetimeSeconds: number;
}
export type JwsVerification<C extends JwsClaims, K extends JwsKey = JwsKey> = {
    readonly ok: true;
    readonly header: JwsHeader;
    readonly claims: C;
    readonly key: K;
} | {
    readonly ok: false;
    readonly code: JwsErrorCode;
};
/** Implemented by @spinajs/authn-keys: compact JWS under the canvas C8 rules (A17). */
export declare abstract class JwsVerifier {
    /**
     * malformed: not three strict base64url parts, or a signature that is not 64 bytes; bad_header: not exactly { alg, kid, typ } with an
     * allowed alg and the expected typ; then unknown_key, bad_signature (checked with the key's own algorithm, never the header's),
     * invalid_claims, wrong_issuer, wrong_audience, expired, not_yet_valid. Bad tokens yield a code; they are never thrown (A17).
     */
    abstract verify<C extends JwsClaims, K extends JwsKey = JwsKey>(token: string, opts: JwsVerifyOptions<C, K>): Promise<JwsVerification<C, K>>;
}
export interface JwsSignOptions {
    readonly typ: string;
    /** One of the signer's own keys; default: its current one. */
    readonly kid?: string;
}
/** Holds its private keys (from a secret reference, or in a KMS); callers never pass or see them. */
export declare abstract class JwsSigner {
    /**
     * Header { alg, kid, typ } in that order, alg from the key; the claims as JSON.stringify writes them, in the caller's key order.
     * EdDSA is deterministic, so the C8 golden tickets come out byte for byte. Throws for a kid it does not hold: a configuration error.
     */
    abstract sign<C extends JwsClaims>(claims: C, opts: JwsSignOptions): Promise<string>;
    /** For a JWKS route (C8 D5) and the verifiers' key lookup. */
    abstract publicKeys(): Promise<readonly JwsKey[]>;
}
```

import type { AuthnContext, IdentityRef } from "./credentials.js";
import type { Ceiling, PrincipalRef } from "./principal.js";
/** Fixed on the key record by its key type; never read from a token header (A17). */
export type KeyAlgorithm = "EdDSA" | "ES256";
/** RFC 8037. */
export interface Ed25519PublicJwk {
    readonly kty: "OKP";
    readonly crv: "Ed25519";
    readonly x: string;
    readonly d?: never;
}
export interface P256PublicJwk {
    readonly kty: "EC";
    readonly crv: "P-256";
    readonly x: string;
    readonly y: string;
    readonly d?: never;
}
/** Public keys only: a JWK with a private part is refused. */
export type PublicJwk = Ed25519PublicJwk | P256PublicJwk;
export type KeyMaterial = {
    readonly alg: "EdDSA";
    readonly jwk: Ed25519PublicJwk;
} | {
    readonly alg: "ES256";
    readonly jwk: P256PublicJwk;
};
export interface KeyOwner extends IdentityRef {
    readonly type: "user" | "service" | "agent";
}
/** A row of authn_key. */
export type KeyRecord = KeyMaterial & {
    /** 1–64 of [A-Za-z0-9._-]; unique. */
    readonly kid: string;
    readonly owner: KeyOwner;
    /** RFC 7638, base64url; unique. */
    readonly thumbprint: string;
    readonly name?: string;
    readonly scope?: Ceiling;
    readonly createdAt: Date;
    readonly createdBy?: PrincipalRef;
    readonly expiresAt?: Date;
    readonly revokedAt?: Date;
    readonly lastUsedAt?: Date;
};
export interface NewKey {
    /** Default: the thumbprint. */
    readonly kid?: string;
    readonly jwk: PublicJwk;
    readonly name?: string;
    readonly scope?: Ceiling;
    readonly expiresAt?: Date;
}
/** The new key's signature over a 'register' challenge issued to the same user (proof of possession). */
export interface KeyProof {
    readonly challengeId: string;
    /** base64url. */
    readonly signature: string;
}
/** A user's key always needs a proof; services and agents are registered without one by an admin route or the CLI, named as `actor`. */
export type KeyRegistration = (NewKey & {
    readonly owner: KeyOwner & {
        readonly type: "user";
    };
    readonly proof: KeyProof;
}) | (NewKey & {
    readonly owner: KeyOwner & {
        readonly type: "service" | "agent";
    };
    readonly actor: PrincipalRef;
});
export type KeyRegisterResult = {
    readonly ok: true;
    readonly key: KeyRecord;
} | {
    readonly ok: false;
    readonly code: "invalid_key" | "invalid_proof" | "duplicate";
};
export type KeyChallengeRequest = {
    readonly purpose: "register";
    readonly owner: KeyOwner & {
        readonly type: "user";
    };
} | {
    readonly purpose: "login";
    readonly kid: string;
};
export interface KeyChallenge {
    /** Names the challenge in a KeyProof or in key-challenge credentials. */
    readonly id: string;
    /** Signed together with the challenge purpose (domain separation), so a login signature never proves a registration. */
    readonly nonce: string;
    readonly expiresAt: Date;
}
export type KeyChallengeResult = {
    readonly ok: true;
    readonly challenge: KeyChallenge;
} | {
    readonly ok: false;
    readonly retryAfterSeconds: number;
};
/** Implemented by @spinajs/authn-keys over authn_key: Ed25519 and P-256 public keys only (A17). */
export declare abstract class KeyRegistry {
    /**
     * POST /user/keys (fresh auth, A13) with a proof; an admin route or authn:key-add for services and agents. Refuses a user's key without
     * a valid proof, whatever the caller. Checks the key, then the proof, then that kid and thumbprint are unused.
     */
    abstract register(reg: KeyRegistration, ctx: AuthnContext): Promise<KeyRegisterResult>;
    /** GET /user/keys; revoked and expired keys included. */
    abstract list(owner: IdentityRef): Promise<readonly KeyRecord[]>;
    /**
     * DELETE /user/keys/:kid. Owner-bound: false unless the key's owner has the same type and id as `owner` (no IDOR); an admin route
     * or the CLI passes the key's owner. A revoked key fails on the next request (A17).
     */
    abstract revoke(owner: IdentityRef, kid: string, ctx: AuthnContext, opts?: {
        readonly actor?: PrincipalRef;
    }): Promise<boolean>;
    /** For verifiers. Returns revoked and expired keys too, so the reason can be audited; accept a key only without revokedAt and before expiresAt. */
    abstract byId(kid: string): Promise<KeyRecord | null>;
    /**
     * POST /auth/keys/challenge (anonymous, login) and the registration proof (an authenticated user).
     * A login challenge first counts against the 'login' flow's IP key; a refusal stores nothing. Every challenge stands alone, stored under its own id (with SecretStore:
     * purpose 'key-challenge:<id>'), so issuing one never cancels another. Single use and short-lived. A login challenge for an unknown kid
     * looks the same and never verifies (A02). A signature covers the purpose too.
     */
    abstract challenge(req: KeyChallengeRequest, ctx: AuthnContext): Promise<KeyChallengeResult>;
}

export interface PasswordPolicy {

    readonly maxLength: number;
    /** ECMAScript regular expression source, when the configured rule has one. */
    readonly pattern?: string;
    /** Copy to show next to the field; never validated against. */

}
/** A policy violation says nothing about the account, so the caller may show it. */
export type PasswordViolation = "too_short" | "too_long" | "rule" | "blocklisted";
/**
 * 'invalid' is the one answer for an unknown or inactive account, a wrong password, and a missing, expired, used or wrong secret
 * (A02, A11, SR-17); the real reason goes to audit only.
 */
export type PasswordRefusal = {
    readonly status: "invalid";
} | {
    readonly status: "throttled";
    readonly retryAfterSeconds: number;
} | {
    readonly status: "rejected";
    readonly violations: readonly PasswordViolation[];
};
export type PasswordChangeResult = {
    readonly status: "changed";
    /** The rotated current session. */
    readonly session: SessionHandle;
} | PasswordRefusal;
export type PasswordResetRequestResult = {
    readonly status: "accepted";
} | {
    readonly status: "throttled";
    readonly retryAfterSeconds: number;
};
export type PasswordResetResult = {
    readonly status: "reset";
    readonly purpose: "reset" | "invite";
} | PasswordRefusal;
export type PasswordSetResult = {
    readonly status: "set";
} | {
    readonly status: "rejected";
    readonly violations: readonly PasswordViolation[];
};
/** Why a password stops working. */
export type PasswordExpiry = {
    /**
     * Admin /authn/admin/users/:uuid/password-reset and the 2.x "expire password" action (A25): revokes every session, incl. those
     * where it is the impersonator, and with authn.local.reset.revokeTokens every token (A30). The caller never receives a secret
     * (A10, SR-10) and applies the target guard.
     */
    readonly cause: "admin";
    readonly actor: PrincipalRef;
    /** Has a reset link mailed to the owner. */
    readonly sendReset?: boolean;
} | {
    /**
     * The authn.local.expiration job: revokes every session, incl. those where it is the impersonator, so the expiry takes effect;
     * keeps every API and agent token (H32); has a reset link mailed to the owner.
     */
    readonly cause: "aged";
};
/** Implemented by @spinajs/authn-local. Passwords are never trimmed and must be 8–128 characters wherever they are set (A04, SR-14). */
export declare abstract class PasswordService {
    /**
     * PATCH /user/password. Checks the policy, then counts the attempt in the 'reauth' throttle and verifies oldPassword inline
     * as fresh auth (A13, SR-20). `session` is the caller's session and must be an active session of `id`; otherwise 'invalid' (fails closed).
     * Revokes every other session of `id`, incl. those where it is the impersonator, rotates `session`
     * and keeps every API and agent token (A30, H32).
     */
    abstract change(id: IdentityRef, oldPassword: string, newPassword: string, session: SessionHandle, ctx: AuthnContext): Promise<PasswordChangeResult>;
    /**
     * POST /auth/password/reset-request. Counts the attempt in the 'reset' flow (IP and account keys) before anything else, then answers
     * at once with the same result whether or not the account exists (A11, SR-04). Returns no secret: it is issued and rendered when the
     * mail is sent (SR-09).
     */
    abstract requestReset(login: string, ctx: AuthnContext): Promise<PasswordResetRequestResult>;
    /**
     * POST /auth/password/reset. Counts the attempt against the 'reset' flow's IP key only (the account key would refuse a link redeemed
     * soon after its request), then checks the policy, then the secret, so a violation tells nothing about the account and does not use
     * up the secret. Single use; takes the same time for an unknown login (A02). Accepts purpose 'reset', or 'invite' while the identity
     * has a pending invite (A32). Revokes every session of the identity, incl. those where it is the impersonator, and with
     * authn.local.reset.revokeTokens every token (A11, A30).
     */
    abstract redeemReset(login: string, secret: string, newPassword: string, ctx: AuthnContext): Promise<PasswordResetResult>;
    /** Issues a secret of purpose 'invite' and has it mailed; returns no secret (SR-09). users marks the identity invite-pending. */

        readonly actor?: PrincipalRef;
    }): Promise<void>;
    /** GET /auth/password/policy. */
    abstract policy(): PasswordPolicy;
    /** Length, rule and blocklist, as every method that sets a password applies them (A04); empty: acceptable. Lets users check before it creates an identity. */
    abstract check(password: string): Promise<readonly PasswordViolation[]>;

     * Sets a password without the old one: users' create and the CLI. Applies `check` (A04). Not a voluntary change: revokes every session

        readonly actor?: PrincipalRef;

import type { IdentityRef } from "./credentials.js";
import type { AuthnFacts } from "./principal.js";
export interface SessionHandle {
    /** Raw id; lives only in memory and the cookie. */
    readonly id: string;
    readonly state: "pending" | "active";
}
/** Implemented by @spinajs/authn-session. */
export declare abstract class SessionIssuer {
    /**
     * With `pending`: a pre-MFA session that yields no principal and expires after authn.session.pendingTtlSeconds (A09).
     * Impersonation start and end each issue a NEW session (start: for the target, with facts.impersonator, stored as its
     * Impersonator; end: for the impersonator) and revoke the old one, so the id rotates and `set` never switches identities.
     */
    abstract issue(identity: IdentityRef, facts: AuthnFacts, opts?: {
        pending?: {
            steps: readonly string[];
        };
    }): Promise<SessionHandle>;
    /**
     * Looks the session up by sha256(rawId); only a uuid v4 input falls back to a raw lookup, so a 2.x id is migrated
     * on first use (A07) and a stored hash never works as an id (SR-18).
     */
    abstract restore(rawId: string): Promise<{
        handle: SessionHandle;
        identity: IdentityRef;
        facts: AuthnFacts;
        data: ReadonlyMap<string, unknown>;
    } | null>;
    /** New id, same data (fixation). */
    abstract rotate(h: SessionHandle): Promise<SessionHandle>;
    /** pending → active, aal2; rotates. Refuses an active session: a re-authentication never changes the facts of a live session. */
    abstract elevate(h: SessionHandle, facts: Partial<AuthnFacts>): Promise<SessionHandle>;
    /** Holds ActiveRole and the impersonation block. Never changes the subject, impersonator or state: the issuer refuses its reserved keys. */
    abstract set(h: SessionHandle, key: string, value: unknown): Promise<void>;
    abstract revoke(h: SessionHandle): Promise<void>;
    /** Returns the count. includeImpersonations also ends sessions in which the subject is the impersonator (SR-06, A08, A20). */
    abstract revokeAll(subject: IdentityRef, opts?: {
        except?: SessionHandle;
        includeImpersonations?: boolean;
    }): Promise<number>;
    /**
     * Serves DELETE /user/sessions/:handle (§22.11). Owner-bound: ends only a session whose subject is `subject` (not one it
     * impersonates from) and whose listed handle equals `handle`; false otherwise (no IDOR). `handle` is the hashed handle of `list`.
     */
    abstract revokeHandle(subject: IdentityRef, handle: string): Promise<boolean>;
    /**
     * `current` marks the row of that handle.
     * Includes sessions in which someone impersonates `subject`: they act as it, so it may see and revoke them,
     * and it sees the impersonator's ip and userAgent.
     * Each `handle` is a one-way value derived from the storage key with domain separation, never the storage key or the raw id;
     * it equals AuthnFacts.sessionHandle and AuthnEvent.session.
     */
    abstract list(subject: IdentityRef, opts?: {
        current?: SessionHandle;
    }): Promise<readonly {
        handle: string;
        createdAt: Date;
        lastSeenAt: Date;
        ip?: string;
        userAgent?: string;
        current?: boolean;
    }[]>;

import type { AuthnContext, AuthnFailure, IdentityRef } from "./credentials.js";
import type { Ceiling, PrincipalRef, ResourceRef } from "./principal.js";
export type TokenKind = "pat" | "agent" | "service";
/**
 * The Scope column: the ceiling limits a token sets beyond its roles. Roles have their own option, the expiry is the token's own,
 * and an agent token's ceiling reason is always 'out_of_scope' (A15), so none of them lives here.
 */
export interface TokenScope {
    readonly permissions?: readonly string[];
    readonly resources?: readonly ResourceRef[];
    /** True exactly when kind is 'agent'. */
    readonly agent?: boolean;
}
export type TokenCreateOptions = {
    readonly name: string;
    /** Mandatory; at most authn.token.maxTtlDays ahead, for an agent token at most authn.token.agentMaxTtlMinutes (A14, A15). */
    readonly expiresAt: Date;
} & ({
    readonly kind: "pat" | "service";
    /** Become ceiling.roles. */
    readonly roles?: readonly string[];
    /** 2.x Profile; authn-token checks it against the profiles the owner may pin a token to. */
    readonly profile?: string;
    readonly scope?: TokenScope & {
        readonly agent?: false;
    };
} | {
    readonly kind: "agent";
    /** An agent acts with its owner's rights narrowed by its scope: roles and a profile are refused (invalid_scope). */
    readonly roles?: never;
    readonly profile?: never;
    readonly scope: TokenScope & {
        readonly agent: true;
    };
});
/** Never holds the plaintext or its hash. */
export interface TokenRecord {
    /** rbac_access_tokens.Uuid, never the row id. */
    readonly id: string;
    /** For an agent token, the identity it acts for. */
    readonly owner: IdentityRef;
    readonly name: string;
    readonly kind: TokenKind;
    readonly roles?: readonly string[];
    /** 2.x Profile: the one role the token is pinned to (§18.3(e)); a profile never widens. */
    readonly profile?: string;
    readonly scope?: TokenScope;
    readonly createdAt: Date;
    readonly expiresAt: Date;
    readonly lastUsedAt?: Date;
    readonly lastUsedIp?: string;
}
export type TokenCreateResult = {
    readonly ok: true;
    /** Returned once; never stored or logged (A14, SR-08). */
    readonly plaintext: string;
    readonly record: TokenRecord;
} | {
    readonly ok: false;
    readonly code: "invalid_expiry" | "invalid_scope";
};
/**
 * A StrategyOutcome; TokenStrategy returns it with `profile` copied to attributes.profile. ceiling.roles are the token's roles. An agent token yields identity
 * { type: 'agent', id: token id }, delegation 'inherit' on behalf of the owner and a ceiling with reason 'out_of_scope' and no roles (A15).
 */
export type TokenValidation = {
    readonly ok: true;
    readonly identity: IdentityRef;
    readonly method: "token";
    readonly ceiling?: Ceiling;
    readonly delegation?: never;
    /** token.profile; reaches the principal as attributes.profile through StrategyOutcome.attributes and principalOf (§18.3(e)). */
    readonly profile?: string;
    readonly token: TokenRecord;
} | {
    readonly ok: true;
    readonly identity: {
        readonly type: "agent";
        readonly id: string;
    };
    readonly method: "token";
    readonly ceiling: Ceiling & {
        readonly roles?: never;
        readonly reason: "out_of_scope";
    };
    readonly delegation: {
        readonly onBehalfOf: IdentityRef;
        readonly mode: "inherit";
    };
    readonly profile?: never;
    readonly token: TokenRecord;
} | {
    readonly ok: false;
    readonly reason: AuthnFailure;
};
/** Implemented by @spinajs/authn-token over rbac_access_tokens; tokens are stored as SHA-256 only. */
export declare abstract class TokenService {
    /**
     * POST /user/tokens; the route requires fresh auth (A13). 'invalid_scope': kind 'agent' without scope.agent, or the reverse, or an
     * agent token with roles or a profile; invalid_expiry: not after ctx.now, or beyond the cap.
     */
    abstract create(owner: IdentityRef, opts: TokenCreateOptions, ctx: AuthnContext): Promise<TokenCreateResult>;
    /** GET /user/tokens: live and expired tokens; revoked ones are left out. */
    abstract list(owner: IdentityRef): Promise<readonly TokenRecord[]>;
    /**
     * DELETE /user/tokens/:uuid. Owner-bound: false unless a live token with that id has an owner of the same type and id as `owner`
     * (no IDOR). The admin route passes the target as owner.
     */
    abstract revoke(owner: IdentityRef, id: string, ctx: AuthnContext, opts?: {
        readonly actor?: PrincipalRef;
    }): Promise<boolean>;
    /** Password reset and admin credential reset (A11, A25); agent tokens acting for `owner` included. Returns the count. */
    abstract revokeAllFor(owner: IdentityRef, ctx: AuthnContext, opts?: {
        readonly actor?: PrincipalRef;
    }): Promise<number>;
    /** Hash lookup; checks revocation, expiry and the owner's status on every use (A14, A15). Bad tokens yield a failure; they are never thrown. */
    abstract validate(raw: string, ctx: AuthnContext): Promise<TokenValidation>;
}
