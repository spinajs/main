import { AuthorizationModel } from "@spinajs/authz";
import type { DecisionRequest, EvaluationContext, ModelCapabilities, ModelSummary, PermissionDef, PermissionRegistry, Principal, ResourceRef, ResourceType, Vote } from "@spinajs/authz";
import type { IContainer } from "@spinajs/di";
export declare const FS_INHERIT_MODEL = "fs-inherit";
/**
 * `fs-inherit` (design §13): permits `fs:<op>` on a file or directory of a filesystem checked in this model's
 * domain when the principal holds `<parent type>:<action>` on one of the path's domain parents (FsPathResolver,
 * nearest first), where `authz.fs.<name>.inherit` maps `fs:<op>`, or a permission that implies it, to `<action>`.
 * A parent decision counts only when it is allowed with a TRUE residual: the parent row is not at hand to settle
 * a condition. It never denies; without a mapping, parents or a parent permit it abstains.
 */
export declare class FsInheritModel extends AuthorizationModel {
    /** The domain whose filesystems this instance serves; not `ctx.domain`, since a child domain may boot it first. */
    readonly domain: string;
    /** Where the Configuration and the FsPathResolvers are resolved. */
    protected readonly container: IContainer;
    readonly name = "fs-inherit";
    readonly capabilities: ModelCapabilities;
    private registry;
    private maps;
    constructor(/** The domain whose filesystems this instance serves; not `ctx.domain`, since a child domain may boot it first. */
    domain: string, /** Where the Configuration and the FsPathResolvers are resolved. */
    container?: IContainer);
    init(ctx: {
        readonly permissions: PermissionRegistry;
    }): void;
    applies(permission: PermissionDef, type: ResourceType): boolean;
    evaluate(req: DecisionRequest, ctx: EvaluationContext): Promise<Vote>;
    /**
     * Every permission the model may grant at `at`, conditional: it depends on the path's parents. So `covers` asks
     * the holder to hold each of them unconditionally (fail closed). At `$global` the filesystems of the domain, at a
     * file or directory those of its filesystem, elsewhere none.
     */
    summarize(_principal: Principal, at: ResourceRef, ctx: EvaluationContext): Promise<ModelSummary>;
    /** The actions `map` gives `permission`: its own entry first, then those of the permissions that imply it. */
    private actions;
}

## src/refs/paths.ts

```ts
/**
 * The path a filesystem's ref ids use (design §13): '/' separators, no '.' or '..' segments, no leading or
 * trailing '/', '' for the root. An absolute path inside `basePath` becomes relative to it; any other rooted
 * path without a drive (`/x`, `\x`) is relative to the filesystem root, as fsNative.resolvePath reads it. A
 * driveless rooted path is never inside a drive or UNC base path: on Windows `\data\up\x` with the base
 * `C:\data\up` gives `data/up/x`, while fsNative would read the base itself on the current drive.
 *
 * The guard hands this normalised path, not the caller's, to the raw provider, so the path it checks is the
 * path the provider touches, also on providers that use keys verbatim (S3: `/x` and `x` would be two keys).
 *
 * @throws InvalidArgument for a path that climbs above the root, holds a NUL byte, is a drive or UNC path
 * outside `basePath`, or has a segment holding ':' or ending in '.' or a space
 */
export declare function normalizeFsPath(path: string, basePath?: string): string;

// values
const DRIVE = /^([A-Za-z]):/;
const UNC = /^\/\/[^/]/;
const ALIAS_SEGMENT = /:|[. ]$/;
```
